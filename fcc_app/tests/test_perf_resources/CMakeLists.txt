cmake_minimum_required(VERSION 3.20.0)
project(test_perf_resources C)
enable_testing()
add_executable(perf_resources_tests src/main.c ../../src/diagnostics/perf_resources.c)
target_include_directories(perf_resources_tests PRIVATE include ../common/include ../../src)
target_compile_features(perf_resources_tests PRIVATE c_std_11)
target_compile_options(perf_resources_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
foreach(test_name IN ITEMS test_values test_query_failures test_stack_minimum)
    add_test(NAME ${test_name} COMMAND perf_resources_tests ${test_name})
endforeach()
